// CUIDA+ Pulseira · firmware v1.4.2
#include <WiFi.h>
#include <MAX30102.h>

const int PANIC_PIN = 4;
const int PULSE_PIN = 34;
MAX30102 sensor(PULSE_PIN);

void setup() {
  Serial.begin(115200);
  sensor.begin();
  connectToCloud();
}

void loop() {
  publishReadings(sensor.readPulse());
  delay(30000);
}
